! ============================================================================
!  SW1 · Cisco vIOS-L2 (IOS 15.2)
!  Funcion: switch de la LAN y agente SNMP de solo lectura
!  Laboratorio: DHCP, SNMP y Zabbix
!  Construido a partir del running-config. Sustituye <...> por tus claves.
! ============================================================================
enable
configure terminal
hostname SW1
no ip domain-lookup
ip domain-name red.local

! --- Acceso administrativo base ---
enable secret <CLAVE_ENABLE>
username Admin privilege 15 secret <CLAVE_ADMIN>
crypto key generate rsa modulus 2048
ip ssh version 2

line con 0
 login local
exit
line vty 0 4
 login local
 transport input ssh
exit

! --- 1. Direccionamiento IP de gestion (VLAN 1) ---
interface Vlan1
 ip address 10.7.1.2 255.255.255.0
 no shutdown
exit
ip default-gateway 10.7.1.1

! --- 3. SNMP de solo lectura ---
snmp-server community ZBX_RO RO
snmp-server location Laboratorio

banner motd ^CLisbeth Gomez 2025-0701^C
end
write memory
